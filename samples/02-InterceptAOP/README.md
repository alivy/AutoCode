# 02 - InterceptAOP：编译时 AOP 替代动态代理

> 展示：`[AutoIntercept]` 如何在编译时生成方法拦截管线，替代 Castle DynamicProxy。

## Before：Castle DynamicProxy 方式（~200 行）

```csharp
// ═══════════════════════════════════════════════════════
// 1. 安装 NuGet 包: Castle.Core (400KB+) + Autofac.Extras.DynamicProxy
// ═══════════════════════════════════════════════════════

// 2. 手写拦截器 (~80行)
public class LoggingInterceptor : IInterceptor
{
    private readonly ILogger _logger;
    public LoggingInterceptor(ILogger logger) => _logger = logger;

    public void Intercept(IInvocation invocation)
    {
        var sw = Stopwatch.StartNew();
        _logger.LogInformation("{Method} 开始, 参数: {Args}",
            invocation.Method.Name,
            string.Join(", ", invocation.Arguments));
        try
        {
            invocation.Proceed();
            // 处理异步
            if (invocation.ReturnValue is Task task)
            {
                task.ContinueWith(t =>
                {
                    if (t.IsFaulted)
                        _logger.LogError(t.Exception, "{Method} 异常", invocation.Method.Name);
                    else
                        _logger.LogInformation("{Method} 完成, 耗时 {Ms}ms",
                            invocation.Method.Name, sw.ElapsedMilliseconds);
                });
            }
            else
            {
                _logger.LogInformation("{Method} 完成, 耗时 {Ms}ms",
                    invocation.Method.Name, sw.ElapsedMilliseconds);
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "{Method} 异常, 耗时 {Ms}ms",
                invocation.Method.Name, sw.ElapsedMilliseconds);
            throw;
        }
    }
}

// 3. 手写缓存拦截器 (~50行)
public class CacheInterceptor : IInterceptor
{
    private readonly IMemoryCache _cache;
    public CacheInterceptor(IMemoryCache cache) => _cache = cache;

    public void Intercept(IInvocation invocation)
    {
        var key = $"{invocation.Method.DeclaringType?.Name}.{invocation.Method.Name}:" +
                  string.Join(":", invocation.Arguments);
        if (_cache.TryGetValue(key, out object? cached))
        {
            invocation.ReturnValue = cached;
            return;
        }
        invocation.Proceed();
        if (invocation.ReturnValue != null)
            _cache.Set(key, invocation.ReturnValue, TimeSpan.FromMinutes(5));
    }
}

// 4. 手写重试拦截器 (~40行)
public class RetryInterceptor : IInterceptor
{
    public void Intercept(IInvocation invocation)
    {
        for (int i = 1; ; i++)
        {
            try { invocation.Proceed(); return; }
            catch when (i < 3) { Thread.Sleep(i * 100); }
        }
    }
}

// 5. DI 配置 (~30行)
var builder = new ContainerBuilder();
builder.RegisterType<OrderService>()
    .As<IOrderService>()
    .EnableInterfaceInterceptors()
    .InterceptedBy(typeof(LoggingInterceptor))
    .InterceptedBy(typeof(CacheInterceptor))
    .InterceptedBy(typeof(RetryInterceptor));
builder.RegisterType<LoggingInterceptor>();
builder.RegisterType<CacheInterceptor>();
builder.RegisterType<RetryInterceptor>();
```

**问题**：
- ❌ 运行时 Reflection.Emit，启动慢
- ❌ 代理类不可见，无法 F12 调试
- ❌ NativeAOT 不兼容
- ❌ Castle.Core 400KB+ 额外依赖
- ❌ 异步方法处理复杂且容易出 Bug

---

## After：AutoCode [AutoIntercept] 方式（~5 行标记）

```csharp
// ═══════════════════════════════════════════════════════
// 唯一改动：在类上加一个特性
// ═══════════════════════════════════════════════════════
using AutoCode.Model;

[AutoIntercept(
    InterceptType.Log | InterceptType.Cache | InterceptType.Retry | InterceptType.Metrics,
    CacheDurationSeconds = 60,
    MaxRetryCount = 3)]
public class OrderService : IOrderService
{
    public async Task<Order?> GetByIdAsync(int id)
    {
        return await _db.Orders.FindAsync(id);  // 纯业务代码
    }

    public async Task<bool> PayAsync(int orderId)
    {
        // 可能超时的操作 → 自动重试 3 次（指数退避）
        return await _paymentGateway.ChargeAsync(orderId);
    }
}
```

**编译后自动生成 `InterceptedOrderService.g.cs`**：

```csharp
// <auto-generated/>  ← 编译时生成，可 F12 跳转调试
public sealed class InterceptedOrderService : IOrderService
{
    private readonly IOrderService _inner;
    private readonly ILogger<InterceptedOrderService> _logger;
    private readonly IMemoryCache _cache;

    public async Task<Order?> GetByIdAsync(int id)
    {
        // ① Cache - Before（命中短路）
        var __cacheKey = $"OrderService.GetByIdAsync:{id}";
        if (_cache.TryGetValue(__cacheKey, out Order? __cached))
            return __cached!;

        // ② Log - Before
        _logger.LogInformation("GetByIdAsync 开始, id={id}", id);
        var __sw = Stopwatch.StartNew();

        // ③ Retry（指数退避）
        for (int __attempt = 1; ; __attempt++)
        {
            try
            {
                var __result = await _inner.GetByIdAsync(id);
                // ④ Log - After
                _logger.LogInformation("GetByIdAsync 完成, 耗时 {Elapsed}ms", __sw.ElapsedMilliseconds);
                // ⑤ Metrics - After
                _duration.Record(__sw.Elapsed.TotalMilliseconds);
                // ⑥ Cache - After
                _cache.Set(__cacheKey, __result, TimeSpan.FromSeconds(60));
                return __result;
            }
            catch (Exception __ex) when (__attempt < 3)
            {
                _logger.LogWarning(__ex, "GetByIdAsync 第{Attempt}次失败，准备重试", __attempt);
                await Task.Delay(__attempt * 200);
            }
        }
    }
}
```

## 对比总结

| 维度 | Castle DynamicProxy | AutoCode [AutoIntercept] |
|------|--------------------|-----------------------|
| 实现时机 | 运行时 Reflection.Emit | **编译时 Source Generator** |
| 额外依赖 | Castle.Core 400KB+ | **零依赖** |
| 可调试性 | ❌ 代理类不可见 | **✅ F12 直接跳转 .g.cs** |
| NativeAOT | ❌ 不兼容 | **✅ 完全兼容** |
| 启动速度 | 慢（动态生成类型） | **即时（编译完即可用）** |
| 异步处理 | 复杂（需手动 ContinueWith） | **自动（生成 async/await）** |
| 类型安全 | 运行时才发现错误 | **编译期检查** |
| 代码量 | ~200 行配置 + 拦截器 | **1 个 Attribute** |
| 运行时开销 | 有（拦截链调用） | **零（就是普通方法调用）** |
